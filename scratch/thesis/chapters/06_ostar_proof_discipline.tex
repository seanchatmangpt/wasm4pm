\begin{equation}
\operatorname{ind}(\slashed{D}_E) = \int_X \operatorname{ch}(E) \wedge \widehat{A}(X) \equiv \mathcal{H}_{BLAKE3} \pmod{\mathbb{Z}}
\end{equation}
